## 合成系统需求分析

### 核心玩法设计（基于DESIGN_V8_DEEP.md）
1. **合成流程**
   - 相同植物相邻放置可触发合成
   - 合成产生更高级植物（🌱 → 🌿 → 🌳 → 🎋 → 🌴 → 🌲）
   - 合成动画和特效

2. **合成规则**
   - 基础合成：2个相同等级植物 → 1个更高等级植物
   - 连击合成：连续合成触发奖励倍率
   - 特殊合成：特定组合产生稀有植物

3. **合成效果**
   - 金币/经验奖励
   - 成就解锁
   - 图鉴收集

## 技术方案选择

### 方案一：渐进式扩展（推荐）
**优点**:
- 充分利用现有代码基础
- 风险可控，可快速验证
- 保持向后兼容

**实现路径**:
1. 在现有GameState类扩展合成状态
2. 在现有Canvas渲染层添加合成特效
3. 扩展交互系统支持拖拽合并

### 方案二：混合渲染方案
**技术栈**: Canvas（核心渲染）+ DOM Overlay（合成交互）
**优点**:
- 复杂交互用DOM实现更简单
- 保持Canvas渲染性能
- 合成特效可CSS实现

### 方案三：完整重构
**不推荐**: 时间成本高，风险大

      3: { name: '大树', emoji: '🌳', baseValue: 50 },
      4: { name: '竹子', emoji: '🎋', baseValue: 100 },
      5: { name: '椰树', emoji: '🌴', baseValue: 200 },
      6: { name: '松树', emoji: '🌲', baseValue: 400 }
    };
  }
  
  // 合成记录方法
  recordCombo(sourcePlants, resultPlant, timestamp) {
    this.combos.push({
      sources: sourcePlants,
      result: resultPlant,
      timestamp: timestamp,
      streak: this.comboStreak,
      multiplier: this.comboMultiplier
    });
    
    // 更新连击状态
    this.updateComboStreak();
  }
  
  updateComboStreak() {
    // 连击逻辑实现
    const now = Date.now();
    const lastCombo = this.combos[this.combos.length - 1];
    
    if (lastCombo && (now - lastCombo.timestamp) < 5000) {
      this.comboStreak++;
      this.comboMultiplier = Math.min(3.0, 1.0 + this.comboStreak * 0.2);
    } else {
      this.comboStreak = 0;
      this.comboMultiplier = 1.0;
    }
  }
}

### 2. 交互系统扩展
```javascript
// 合成交互管理器
class SynthesisManager {
  constructor(gameInstance) {
    this.game = gameInstance;
    this.selectedPlant = null;
    this.comboMode = false;
    this.dragStartPos = null;
  }
  
  // 处理合成交互
  handleSynthesisInteraction(x, y, plantId) {
    if (!this.comboMode) {
      // 进入合成模式
      this.selectedPlant = plantId;
      this.comboMode = true;
      this.showSynthesisGuide();
      return;
    }
    
    // 执行合成检查
    if (this.canSynthesize(this.selectedPlant, plantId)) {
      this.performSynthesis(this.selectedPlant, plantId);
      this.comboMode = false;
      this.selectedPlant = null;
    } else {
      // 合成失败提示
      this.showSynthesisError();
      this.comboMode = false;
      this.selectedPlant = null;
    }
  }
  
  canSynthesize(plantA, plantB) {
    // 检查是否可合成
    const flowerA = this.game.flowers[plantA];
    const flowerB = this.game.flowers[plantB];
    
    // 条件1: 相同植物类型
    if (flowerA.type !== flowerB.type) return false;
    
    // 条件2: 相邻位置（优化后可放宽）
    const posA = this.getPlotPosition(plantA);
    const posB = this.getPlotPosition(plantB);
    const distance = Math.abs(posA.row - posB.row) + Math.abs(posA.col - posB.col);
    
    return distance <= 2; // 允许对角相邻
  }
  
  performSynthesis(plantA, plantB) {
    // 执行合成逻辑
    const flowerA = this.game.flowers[plantA];
    const flowerB = this.game.flowers[plantB];
    
    // 1. 计算合成结果
    const resultType = this.getSynthesisResult(flowerA, flowerB);
    const resultLevel = flowerA.level + 1;
    
    // 2. 创建结果植物（在plantA位置）
    const resultPlant = {
      id: plantA,
      type: resultType,
      level: resultLevel,
      growthStartTime: Date.now(),
      synthesisTime: Date.now()
    };
    
    // 3. 删除被合成的植物（plantB）
    delete this.game.flowers[plantB];
    
    // 4. 更新植物状态
    this.game.flowers[plantA] = resultPlant;
    
    // 5. 记录合成
    this.game.state.recordCombo([plantA, plantB], resultPlant, Date.now());
    
    // 6. 触发奖励
    this.giveSynthesisReward(resultPlant);
    
    // 7. 播放合成动画
    this.playSynthesisAnimation(plantA, plantB, resultPlant);
  }
}
```

// Canvas合成特效渲染
class SynthesisRenderer {
  constructor(ctx) {
    this.ctx = ctx;
    this.particles = [];
  }
  
  // 渲染合成特效
  renderSynthesisEffect(sourceX, sourceY, targetX, targetY, resultType) {
    // 1. 拖拽连接线
    this.renderConnectionLine(sourceX, sourceY, targetX, targetY);
    
    // 2. 粒子特效
    this.renderParticleEffect(sourceX, sourceY, targetX, targetY);
    
    // 3. 合成闪光
    this.renderMergeFlash(targetX, targetY);
    
    // 4. 结果展示
    this.renderResultEffect(targetX, targetY, resultType);
  }
  
  renderConnectionLine(x1, y1, x2, y2) {
    this.ctx.save();
    this.ctx.strokeStyle = 'rgba(255, 215, 0, 0.7)';
    this.ctx.lineWidth = 3;
    this.ctx.setLineDash([5, 5]);
    this.ctx.beginPath();
    this.ctx.moveTo(x1, y1);
    this.ctx.lineTo(x2, y2);
    this.ctx.stroke();
    this.ctx.restore();
  }
  
  renderMergeFlash(x, y) {
    // 闪光动画实现
    const gradient = this.ctx.createRadialGradient(x, y, 0, x, y, 50);
    gradient.addColorStop(0, 'rgba(255, 255, 255, 0.8)');
    gradient.addColorStop(0.5, 'rgba(255, 215, 0, 0.5)');
    gradient.addColorStop(1, 'rgba(255, 215, 0, 0)');
    
    this.ctx.save();
    this.ctx.fillStyle = gradient;
    this.ctx.beginPath();
    this.ctx.arc(x, y, 50, 0, Math.PI * 2);
    this.ctx.fill();
    this.ctx.restore();
  }
}

        <span class="combo-count">0</span>
        <span class="combo-multiplier">×1.0</span>
      </div>
    `;
    this.container.appendChild(this.uiElements.comboCounter);
  }
  
  updateComboDisplay(streak, multiplier) {
    if (this.uiElements.comboCounter) {
      const countEl = this.uiElements.comboCounter.querySelector('.combo-count');
      const multiEl = this.uiElements.comboCounter.querySelector('.combo-multiplier');
      
      if (countEl) countEl.textContent = streak;
      if (multiEl) multiEl.textContent = `×${multiplier.toFixed(1)}`;
      
      // 连击动画
      if (streak > 0) {
        this.uiElements.comboCounter.classList.add('active');
        setTimeout(() => {
          this.uiElements.comboCounter.classList.remove('active');
        }, 500);
      }
    }
  }
}
